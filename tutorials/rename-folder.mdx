    გამოსულ მენიუში დააჭირეთ Rename ოფციას.
  </Step>

  <Step title="შეიყვანეთ ახალი სახელი">
    წაშალეთ ძველი სახელი, შეიყვანეთ ახალი და დააჭირეთ Save.
  </Step>
</Steps>
